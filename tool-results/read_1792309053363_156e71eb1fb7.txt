   113→   113→  loading,
   114→   114→  onBack,

   116→   116→  onOpenFolder,
   117→   117→}: AdminPanelProps) {
   118→   118→  const { toast } = useToast();
   119→   119→
   120→   120→  // Category state

   122→   122→  const [newCatSlug, setNewCatSlug] = useState('');
   123→   123→

   128→   128→  const [formSubmitting, setFormSubmitting] = useState(false);
   129→   129→
   130→   130→  // Delete state

   132→   132→  const [deleting, setDeleting] = useState(false);
   133→   133→
   134→   134→  // Auto-generate slug from title

   139→   139→      slug: title

   147→   147→
   148→   148→  const handleCategorySlugChange = (name: string) => {

   151→   151→        .toLowerCase()
   152→   152→        .replace(/\s+/g, '-')

   155→   155→        .replace(/^-|-$/g, '')

   209→   209→      toast({ title: 'خطا', description: 'خطا در آپلود عکس', variant: 'destructive' });

   219→   219→      return;
   220→   220→    }

   223→   223→      const url = editingMovie ? `/api/movies/${editingMovie.id}` : '/api/movies';
   224→   224→      const method = editingMovie ? 'PUT' : 'POST';
   225→   225→      const body = {
   226→   226→        ...formData,
   227→   227→        year: formData.year ? parseInt(formData.year) : null,
   228→   228→        rating: formData.rating ? parseFloat(formData.rating) : null,
   229→   229→      };

   231→   231→        method,
   232→   232→        headers: { 'Content-Type': 'application/json' },
   233→   233→        body: JSON.stringify(body),
   234→   234→      });
   235→   235→      if (!res.ok) {
   236→   236→        const data = await res.json().catch(() => ({}));
   237→   237→        throw new Error(data.error || 'خطا در ذخیره فیلم');
   238→   238→      }
   239→   239→      toast({ title: editingMovie ? 'فیلم ویرایش شد' : 'فیلم اضافه شد' });
   240→   240→      setShowMovieForm(false);
   241→   241→      onDataChange();

   244→   244→        title: 'خطا',
   245→   245→        description: err instanceof Error ? err.message : 'خطای ناشناخته',
   246→   246→        variant: 'destructive',
   247→   247→      });

   249→   249→      setFormSubmitting(false);
   250→   250→    }
   251→   251→  };

   253→   253→  // Delete movie/category
   254→   254→  const handleDelete = async () => {

   256→   256→    setDeleting(true);

   258→   258→      const url = deleteTarget.type === 'movie'
   259→   259→        ? `/api/movies/${deleteTarget.id}`

   265→   265→      onDataChange();

   267→   267→      toast({ title: 'خطا', description: 'عملیات حذف ناموفق بود', variant: 'destructive' });

   269→   269→      setDeleting(false);
   270→   270→    }
   271→   271→  };

   277→   277→      const res = await fetch('/api/categories', {
   278→   278→        method: 'POST',

   281→   281→      });
   282→   282→      if (!res.ok) {
   283→   283→        const data = await res.json().catch(() => ({}));
   284→   284→        throw new Error(data.error || 'خطا');
   285→   285→      }

   287→   287→      setNewCatName('');
   288→   288→      setNewCatSlug('');
   289→   289→      onDataChange();
   290→   290→    } catch (err) {
   291→   291→      toast({
   292→   292→        title: 'خطا',
   293→   293→        description: err instanceof Error ? err.message : 'خطا در افزودن دسته‌بندی',
   294→   294→        variant: 'destructive',
   295→   295→      });
   296→   296→    }
   297→   297→  };
   298→   298→
   299→   299→  // Toggle category in form
   300→   300→  const toggleCategory = (catId: string) => {

   303→   303→      categoryIds: prev.categoryIds.includes(catId)
   304→   304→        ? prev.categoryIds.filter((id) => id !== catId)

   306→   306→    }));
   307→   307→  };
   308→   308→
   309→   309→  return (
   310→   310→    <div className="max-w-7xl mx-auto px-4 sm:px-6 pb-10">
   311→   311→      {/* Header */}
   312→   312→      <div className="flex items-center gap-3 mb-8">
   313→   313→        <Button
   314→   314→          variant="outline"
   315→   315→          className="border-cinematic-border bg-cinematic-surface text-cinematic-muted hover:bg-cinematic-card hover:text-white"

   317→   317→        >
   318→   318→          <ArrowRight className="h-4 w-4 ml-1" />
   319→   319→          بازگشت به سایت
   320→   320→        </Button>
   321→   321→        <div className="flex items-center gap-2">
   322→   322→          <Film className="h-5 w-5 text-primary" />

   324→   324→        </div>
   325→   325→      </div>
   326→   326→
   327→   327→      {/* Categories Section */}

   329→   329→        className="mb-10"

   336→   336→        <div className="rounded-xl border border-cinematic-border bg-cinematic-card p-4 space-y-4">

   382→   382→                    className="flex items-center justify-between rounded-lg bg-cinematic-surface/50 px-3 py-2"

   401→   401→                  </div>

   408→   408→
   409→   409→      {/* Movies Section */}
   410→   410→      <section>
   411→   411→        <div className="flex items-center justify-between mb-4">

   413→   413→            <Film className="h-5 w-5 text-primary" />
   414→   414→            <h2 className="text-lg font-bold text-white">مدیریت فیلم‌ها</h2>

   416→   416→              {movies.length} فیلم
   417→   417→            </Badge>
   418→   418→          </div>
   419→   419→          <Button
   420→   420→            onClick={openNewMovieForm}
   421→   421→            className="bg-primary hover:bg-primary/90 text-white"
   422→   422→          >
   423→   423→            <Plus className="h-4 w-4 ml-1" />
   424→   424→            افزودن فیلم
   425→   425→          </Button>
   426→   426→        </div>
   427→   427→
   428→   428→        {/* Movies List */}
   429→   429→        <div className="rounded-xl border border-cinematic-border bg-cinematic-card overflow-hidden">
