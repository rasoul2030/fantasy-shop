     4→

    18→} from 'lucide-react';

    22→import { Label } from '@/components/ui/label';
    23→import { Badge } from '@/components/ui/badge';
    24→import { Checkbox } from '@/components/ui/checkbox';
    25→import { Skeleton } from '@/components/ui/skeleton';
    26→import {
    27→  Dialog,

    32→import {

    41→} from '@/components/ui/alert-dialog';
    42→import {
    43→  Select,

    48→} from '@/components/ui/select';
    49→import { useToast } from '@/hooks/use-toast';

    51→import { type Movie, type MovieCategory } from './MovieCard';
    52→
    53→interface Category {
    54→  id: string;
    55→  name: string;
    56→  slug: string;
    57→  type: string;
    58→  _count?: { movies: number };
    59→}
    60→
    61→interface AdminPanelProps {
    62→  categories: Category[];
    63→  movies: Movie[];
    64→  loading: boolean;
    65→  onBack: () => void;
    66→  onDataChange: () => void;
    67→  onOpenFolder: (movieId: string) => void;
    68→}
    69→
    70→interface MovieFormData {
    71→  title: string;
    72→  code: string;
    73→  slug: string;
    74→  description: string;
    75→  posterUrl: string;
    76→  folderPath: string;
    77→  year: string;
    78→  ageRating: string;
    79→  status: string;
    80→  episodeInfo: string;
    81→  subType: string;
    82→  rating: string;
    83→  featured: boolean;
    84→  categoryIds: string[];
    85→}
    86→
    87→const emptyForm: MovieFormData = {
    88→  title: '',
    89→  code: '',
    90→  slug: '',
    91→  description: '',
    92→  posterUrl: '',
    93→  folderPath: '',
    94→  year: '',
    95→  ageRating: '',
    96→  status: 'completed',
    97→  episodeInfo: '',
    98→  subType: '',
    99→  rating: '',
   100→  featured: false,
   101→  categoryIds: [],
   102→};
   103→
   104→const statusLabels: Record<string, string> = {
   105→  airing: 'در حال پخش',
   106→  completed: 'تکمیل شده',
   107→  upcoming: 'آینده',
   108→};
   109→
   110→export default function AdminPanel({
   111→  categories,
   112→  movies,
   113→  loading,
   114→  onBack,
   115→  onDataChange,
   116→  onOpenFolder,
   117→}: AdminPanelProps) {
   118→  const { toast } = useToast();
   119→
   120→  // Category state
   121→  const [newCatName, setNewCatName] = useState('');
   122→  const [newCatSlug, setNewCatSlug] = useState('');
   123→
   124→  // Movie form state
   125→  const [showMovieForm, setShowMovieForm] = useState(false);
   126→  const [editingMovie, setEditingMovie] = useState<Movie | null>(null);
   127→  const [formData, setFormData] = useState<MovieFormData>(emptyForm);
   128→  const [formSubmitting, setFormSubmitting] = useState(false);
   129→
   130→  // Delete state
   131→  const [deleteTarget, setDeleteTarget] = useState<{ type: 'movie' | 'category'; id: string; name: string } | null>(null);
   132→  const [deleting, setDeleting] = useState(false);
   133→
   134→  // Auto-generate slug from title
   135→  const handleTitleChange = (title: string) => {
   136→    setFormData((prev) => ({
   137→      ...prev,
   138→      title,
   139→      slug: title
   140→        .toLowerCase()
   141→        .replace(/\s+/g, '-')
   142→        .replace(/[^\u0600-\u06FFa-z0-9-]/g, '')
   143→        .replace(/-+/g, '-')
   144→        .replace(/^-|-$/g, ''),
   145→    }));
   146→  };
   147→
   148→  const handleCategorySlugChange = (name: string) => {
   149→    setNewCatSlug(
   150→      name
   151→        .toLowerCase()
   152→        .replace(/\s+/g, '-')
   153→        .replace(/[^\u0600-\u06FFa-z0-9-]/g, '')
   154→        .replace(/-+/g, '-')
   155→        .replace(/^-|-$/g, '')
   156→    );
   157→  };
   158→
   159→  // Open form for new movie
   160→  const openNewMovieForm = () => {
   161→    setEditingMovie(null);
   162→    setFormData(emptyForm);
   163→    setShowMovieForm(true);
   164→  };
   165→
   166→  // Open form for editing
   167→  const openEditMovieForm = (movie: Movie) => {
   168→    setEditingMovie(movie);
   169→    setFormData({
   170→      title: movie.title,
   171→      code: movie.code || '',
   172→      slug: movie.slug,
   173→      description: movie.description || '',
   174→      posterUrl: movie.posterUrl || '',
   175→      folderPath: movie.folderPath || '',
   176→      year: movie.year?.toString() || '',
   177→      ageRating: movie.ageRating || '',
   178→      status: movie.status,
   179→      episodeInfo: movie.episodeInfo || '',
   180→      subType: movie.subType || '',
   181→      rating: movie.rating?.toString() || '',
   182→      featured: movie.featured || false,
   183→      categoryIds: movie.categories.map((mc) => mc.categoryId),
   184→    });
   185→    setShowMovieForm(true);
   186→  };
   187→
   188→  // Upload poster image
   189→  const handlePosterUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   190→    const file = e.target.files?.[0];
   191→    if (!file) return;
   192→
   193→    const formDataUpload = new FormData();
   194→    formDataUpload.append('file', file);
   195→
   196→    try {
   197→      const res = await fetch('/api/upload-poster', {
   198→        method: 'POST',
   199→        body: formDataUpload,
   200→      });
   201→      const data = await res.json();
   202→      if (data.url) {
   203→        setFormData((p) => ({ ...p, posterUrl: data.url }));
   204→        toast({ title: 'آپلود موفق', description: 'عکس پوستر آپلود شد' });
   205→      } else {
   206→        toast({ title: 'خطا', description: data.error || 'آپلود ناموفق', variant: 'destructive' });
   207→      }
   208→    } catch {
   209→      toast({ title: 'خطا', description: 'خطا در آپلود عکس', variant: 'destructive' });
   210→    }
   211→    // Reset file input
   212→    e.target.value = '';
   213→  };
   214→
   215→  // Save movie (create or update)
   216→  const handleSaveMovie = async () => {
   217→    if (!formData.title.trim() || !formData.slug.trim()) {
   218→      toast({ title: 'خطا', description: 'عنوان و اسلاگ الزامی است', variant: 'destructive' });
   219→      return;
   220→    }
   221→    setFormSubmitting(true);
   222→    try {
   223→      const url = editingMovie ? `/api/movies/${editingMovie.id}` : '/api/movies';
   224→      const method = editingMovie ? 'PUT' : 'POST';
   225→      const body = {
   226→        ...formData,
   227→        year: formData.year ? parseInt(formData.year) : null,
   228→        rating: formData.rating ? parseFloat(formData.rating) : null,
   229→      };
   230→      const res = await fetch(url, {
   231→        method,
   232→        headers: { 'Content-Type': 'application/json' },
   233→        body: JSON.stringify(body),
   234→      });
   235→      if (!res.ok) {
   236→        const data = await res.json().catch(() => ({}));
   237→        throw new Error(data.error || 'خطا در ذخیره فیلم');
   238→      }
   239→      toast({ title: editingMovie ? 'فیلم ویرایش شد' : 'فیلم اضافه شد' });
   240→      setShowMovieForm(false);
   241→      onDataChange();
   242→    } catch (err) {
   243→      toast({
   244→        title: 'خطا',
   245→        description: err instanceof Error ? err.message : 'خطای ناشناخته',
   246→        variant: 'destructive',
   247→      });
   248→    } finally {
   249→      setFormSubmitting(false);
   250→    }
   251→  };
   252→
   253→  // Delete movie/category
   254→  const handleDelete = async () => {
   255→    if (!deleteTarget) return;
   256→    setDeleting(true);
   257→    try {
   258→      const url = deleteTarget.type === 'movie'
   259→        ? `/api/movies/${deleteTarget.id}`
   260→        : `/api/categories?id=${deleteTarget.id}`;
   261→      const res = await fetch(url, { method: 'DELETE' });
   262→      if (!res.ok) throw new Error('خطا در حذف');
   263→      toast({ title: 'با موفقیت حذف شد' });
   264→      setDeleteTarget(null);
   265→      onDataChange();
   266→    } catch {
   267→      toast({ title: 'خطا', description: 'عملیات حذف ناموفق بود', variant: 'destructive' });
   268→    } finally {
   269→      setDeleting(false);
   270→    }
   271→  };
   272→
   273→  // Add category
   274→  const handleAddCategory = async () => {
   275→    if (!newCatName.trim() || !newCatSlug.trim()) return;
   276→    try {
   277→      const res = await fetch('/api/categories', {
   278→        method: 'POST',
   279→        headers: { 'Content-Type': 'application/json' },
   280→        body: JSON.stringify({ name: newCatName, slug: newCatSlug }),
   281→      });
   282→      if (!res.ok) {
   283→        const data = await res.json().catch(() => ({}));
   284→        throw new Error(data.error || 'خطا');
   285→      }
   286→      toast({ title: 'دسته‌بندی اضافه شد' });
   287→      setNewCatName('');
   288→      setNewCatSlug('');
   289→      onDataChange();
   290→    } catch (err) {
   291→      toast({
   292→        title: 'خطا',
   293→        description: err instanceof Error ? err.message : 'خطا در افزودن دسته‌بندی',
   294→        variant: 'destructive',
   295→      });
   296→    }
   297→  };
   298→
   299→  // Toggle category in form
   300→  const toggleCategory = (catId: string) => {
   301→    setFormData((prev) => ({
   302→      ...prev,
   303→      categoryIds: prev.categoryIds.includes(catId)
   304→        ? prev.categoryIds.filter((id) => id !== catId)
   305→        : [...prev.categoryIds, catId],
   306→    }));
   307→  };
   308→
   309→  return (
   310→    <div className="max-w-7xl mx-auto px-4 sm:px-6 pb-10">
   311→      {/* Header */}
   312→      <div className="flex items-center gap-3 mb-8">
   313→        <Button
   314→          variant="outline"
   315→          className="border-cinematic-border bg-cinematic-surface text-cinematic-muted hover:bg-cinematic-card hover:text-white"
   316→          onClick={onBack}
   317→        >
   318→          <ArrowRight className="h-4 w-4 ml-1" />
   319→          بازگشت به سایت
   320→        </Button>
   321→        <div className="flex items-center gap-2">
   322→          <Film className="h-5 w-5 text-primary" />
   323→          <h1 className="text-xl font-bold text-white">پنل مدیریت</h1>
   324→        </div>
   325→      </div>
   326→
   327→      {/* Categories Section */}
   328→      <section
   329→        className="mb-10"
   330→      >
   331→        <div className="flex items-center gap-2 mb-4">
   332→          <Tag className="h-5 w-5 text-primary" />
   333→          <h2 className="text-lg font-bold text-white">مدیریت دسته‌بندی‌ها</h2>
   334→        </div>
   335→
   336→        <div className="rounded-xl border border-cinematic-border bg-cinematic-card p-4 space-y-4">
   337→          {/* Add Category */}
   338→          <div className="flex flex-col sm:flex-row items-stretch sm:items-end gap-3">
   339→            <div className="flex-1 space-y-1.5">
   340→              <Label className="text-cinematic-muted text-sm">نام دسته‌بندی</Label>
   341→              <Input
   342→                value={newCatName}
   343→                onChange={(e) => {
   344→                  setNewCatName(e.target.value);
   345→                  if (!newCatSlug) handleCategorySlugChange(e.target.value);
   346→                }}
   347→                placeholder="مثلاً: اکشن"
   348→                className="border-cinematic-border bg-cinematic-surface text-white placeholder:text-cinematic-muted/50"
   349→              />
   350→            </div>
   351→            <div className="flex-1 space-y-1.5">
   352→              <Label className="text-cinematic-muted text-sm">اسلاگ</Label>
   353→              <Input
   354→                value={newCatSlug}
   355→                onChange={(e) => setNewCatSlug(e.target.value)}
   356→                placeholder="مثلاً: action"
   357→                dir="ltr"
   358→                className="border-cinematic-border bg-cinematic-surface text-white placeholder:text-cinematic-muted/50"
   359→              />
   360→            </div>
   361→            <Button
   362→              onClick={handleAddCategory}
   363→              disabled={!newCatName.trim() || !newCatSlug.trim()}
   364→              className="bg-primary hover:bg-primary/90 text-white h-10 shrink-0"
   365→            >
   366→              <Plus className="h-4 w-4 ml-1" />
   367→              افزودن
   368→            </Button>
   369→          </div>
   370→
   371→          {/* Category List */}
   372→          <div className="max-h-48 overflow-y-auto space-y-1.5">
   373→            {loading ? (
   374→              Array.from({ length: 4 }).map((_, i) => (
   375→                <Skeleton key={i} className="h-10 w-full rounded-lg bg-cinematic-surface" />
   376→              ))
   377→            ) : (
   378→              <>
   379→                {categories.map((cat) => (
   380→                  <div
   381→                    key={cat.id}
   382→                    className="flex items-center justify-between rounded-lg bg-cinematic-surface/50 px-3 py-2"
   383→                  >
   384→                    <div className="flex items-center gap-2">
   385→                      <span className="text-sm text-white">{cat.name}</span>
   386→                      <Badge variant="outline" className="text-[10px] text-cinematic-muted border-cinematic-border" dir="ltr">
   387→                        {cat.slug}
   388→                      </Badge>
   389→                      <span className="text-[10px] text-cinematic-muted">
   390→                        ({cat._count?.movies || 0} فیلم)
   391→                      </span>
   392→                    </div>
   393→                    <Button
   394→                      variant="ghost"
   395→                      size="icon"
   396→                      className="h-7 w-7 text-red-400 hover:text-red-300 hover:bg-red-500/10"
   397→                      onClick={() => setDeleteTarget({ type: 'category', id: cat.id, name: cat.name })}
   398→                    >
   399→                      <Trash2 className="h-3.5 w-3.5" />
   400→                    </Button>
   401→                  </div>
   402→                ))}
   403→              </>
   404→            )}
   405→          </div>
   406→        </div>
   407→      </section>
   408→
   409→      {/* Movies Section */}
   410→      <section>
   411→        <div className="flex items-center justify-between mb-4">
   412→          <div className="flex items-center gap-2">
   413→            <Film className="h-5 w-5 text-primary" />
   414→            <h2 className="text-lg font-bold text-white">مدیریت فیلم‌ها</h2>
   415→            <Badge variant="outline" className="text-xs text-cinematic-muted border-cinematic-border">
   416→              {movies.length} فیلم
   417→            </Badge>
   418→          </div>
   419→          <Button
   420→            onClick={openNewMovieForm}
   421→            className="bg-primary hover:bg-primary/90 text-white"
   422→          >
   423→            <Plus className="h-4 w-4 ml-1" />
   424→            افزودن فیلم
   425→          </Button>
   426→        </div>
   427→
   428→        {/* Movies List */}
   429→        <div className="rounded-xl border border-cinematic-border bg-cinematic-card overflow-hidden">
   430→          {loading ? (
   431→            <div className="p-4 space-y-3">
   432→              {Array.from({ length: 4 }).map((_, i) => (
   433→                <Skeleton key={i} className="h-16 w-full rounded-lg bg-cinematic-surface" />
   434→              ))}
   435→            </div>
   436→          ) : movies.length === 0 ? (
   437→            <div className="flex flex-col items-center justify-center py-12 text-cinematic-muted">
   438→              <Film className="h-10 w-10 mb-2 opacity-40" />
   439→              <p className="text-sm">فیلمی وجود ندارد</p>
   440→            </div>
   441→          ) : (
   442→            <div className="max-h-[500px] overflow-y-auto">
   443→              {movies.map((movie) => (
   444→                <div
   445→                  key={movie.id}
   446→                  className="flex items-center gap-3 border-b border-cinematic-border last:border-0 px-4 py-3 hover:bg-cinematic-surface/50 transition-colors"
   447→                >
   448→                  {/* Thumbnail */}
   449→                  <div className="shrink-0 h-14 w-10 rounded-lg overflow-hidden bg-cinematic-surface">
   450→                    {movie.posterUrl ? (
   451→                      <img src={movie.posterUrl} alt={movie.title} className="h-full w-full object-cover" />
   452→                    ) : (
   453→                      <div className="flex h-full w-full items-center justify-center text-lg">🎬</div>
   454→                    )}
   455→                  </div>
   456→
   457→                  {/* Info */}
   458→                  <div className="flex-1 min-w-0">
   459→                    <h3 className="text-sm font-medium text-white truncate">{movie.title}</h3>
   460→                    <div className="flex items-center gap-2 mt-0.5">
   461→                      <span className="text-xs text-cinematic-muted">{movie.year || '—'}</span>
   462→                      <Badge className={cn(
   463→                        'text-[10px] px-1 py-0',
   464→                        movie.status === 'airing'
   465→                          ? 'bg-rose-500/20 text-rose-400 border-0'
   466→                          : movie.status === 'completed'
   467→                          ? 'bg-emerald-500/20 text-emerald-400 border-0'
   468→                          : 'bg-amber-500/20 text-amber-400 border-0'
   469→                      )}>
   470→                        {statusLabels[movie.status] || movie.status}
   471→                      </Badge>
   472→                    </div>
   473→                  </div>
   474→
   475→                  {/* Actions */}
   476→                  <div className="flex items-center gap-1 shrink-0">
   477→                    <Button
   478→                      variant="ghost"
   479→                      size="icon"
   480→                      className="h-8 w-8 text-cinematic-muted hover:text-amber-400 hover:bg-amber-500/10"
   481→                      onClick={() => onOpenFolder(movie.id)}
   482→                      title="باز کردن پوشه"
   483→                    >
   484→                      <FolderOpen className="h-4 w-4" />
   485→                    </Button>
   486→                    <Button
   487→                      variant="ghost"
   488→                      size="icon"
   489→                      className="h-8 w-8 text-cinematic-muted hover:text-primary hover:bg-primary/10"
   490→                      onClick={() => openEditMovieForm(movie)}
   491→                      title="ویرایش"
   492→                    >
   493→                      <Edit className="h-4 w-4" />
   494→                    </Button>
   495→                    <Button
   496→                      variant="ghost"
   497→                      size="icon"
   498→                      className="h-8 w-8 text-cinematic-muted hover:text-red-400 hover:bg-red-500/10"
   499→                      onClick={() => setDeleteTarget({ type: 'movie', id: movie.id, name: movie.title })}
   500→                      title="حذف"
   501→                    >
   502→                      <Trash2 className="h-4 w-4" />
   503→                    </Button>
   504→                  </div>
   505→                </div>
   506→              ))}
   507→            </div>
   508→          )}
   509→        </div>
   510→      </section>
   511→
   512→      {/* Movie Form Dialog */}
   513→      <Dialog open={showMovieForm} onOpenChange={(v) => !v && setShowMovieForm(false)}>
   514→        <DialogContent className="max-w-2xl w-[95vw] max-h-[90vh] overflow-y-auto border-cinematic-border bg-cinematic-surface p-0">
   515→          <div className="sticky top-0 z-10 flex items-center justify-between border-b border-cinematic-border bg-cinematic-surface px-6 py-4">
   516→            <DialogHeader>
   517→              <DialogTitle className="text-lg font-bold text-white">
   518→                {editingMovie ? 'ویرایش فیلم' : 'افزودن فیلم جدید'}
   519→              </DialogTitle>
   520→            </DialogHeader>
   521→            <Button
   522→              variant="ghost"
   523→              size="icon"
   524→              className="h-8 w-8 text-cinematic-muted hover:text-white"
   525→              onClick={() => setShowMovieForm(false)}
   526→            >
   527→              <X className="h-4 w-4" />
   528→            </Button>
   529→          </div>
   530→          <div className="p-6 space-y-4">
   531→            {/* Title, Slug & Code */}
   532→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   533→              <div className="space-y-1.5">
   534→                <Label className="text-cinematic-muted text-sm">عنوان *</Label>
   535→                <Input
   536→                  value={formData.title}
   537→                  onChange={(e) => handleTitleChange(e.target.value)}
   538→                  placeholder="عنوان فیلم"
   539→                  className="border-cinematic-border bg-cinematic-card text-white placeholder:text-cinematic-muted/50"
   540→                />
   541→              </div>
   542→              <div className="space-y-1.5">
   543→                <Label className="text-cinematic-muted text-sm">کد</Label>
   544→                <Input
   545→                  value={formData.code}
   546→                  onChange={(e) => setFormData((p) => ({ ...p, code: e.target.value }))}
   547→                  placeholder="کد اختصاصی"
   548→                  dir="ltr"
   549→                  className="border-cinematic-border bg-cinematic-card text-white placeholder:text-cinematic-muted/50 font-mono"
   550→                />
   551→              </div>
   552→              <div className="space-y-1.5">
   553→                <Label className="text-cinematic-muted text-sm">اسلاگ *</Label>
   554→                <Input
   555→                  value={formData.slug}
   556→                  onChange={(e) => setFormData((p) => ({ ...p, slug: e.target.value }))}
